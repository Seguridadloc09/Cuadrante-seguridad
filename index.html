// ============================================================ // CUADRANTE DE SEGURIDAD - Apps Script completo (v2) // Pegar en: Extensiones → Apps Script (borrar TODO lo anterior y pegar este) // // Incluye: // 1) Menú "Cuadrante" en la planilla con: // - Generar francos del mes (usa columnas AK "Ciclo días" y AL "Franco ref.") // - Archivar mes y avanzar al siguiente (guarda copia histórica y limpia) // 2) Registro de última modificación en la celda P1 (lo muestra la web) // ============================================================ // ---------- MENÚ ---------- function onOpen() { try { SpreadsheetApp.getUi().createMenu('Cuadrante') .addItem('Generar francos del mes', 'generarFrancos') .addItem('Archivar mes y avanzar al siguiente', 'archivarYAvanzarMes') .addToUi(); } catch (e) { // fuera del contexto de planilla abierta no hay menú } } // ---------- GENERADOR AUTOMÁTICO DE FRANCOS ---------- // Para cada empleado pone "F" cada X días (columna AK) contando desde la fecha // de referencia (columna AL). Respeta marcas manuales (VAC, TM TN, NT, 9a17, FE): // solo escribe en celdas vacías o que ya tienen F. Los F viejos se regeneran. function generarFrancos() { var ss = SpreadsheetApp.getActive(); var sh = ss.getSheetByName('Hoja1'); if (!sh) return; var anio = parseInt(sh.getRange('E1').getValue(), 10); var mes = parseInt(sh.getRange('G1').getValue(), 10); // 1..12 var dias = new Date(anio, mes, 0).getDate(); var bloques = [[4, 10], [13, 19], [22, 22]]; var DIA_MS = 86400000; var puestos = 0; for (var b = 0; b < bloques.length; b++) { for (var r = bloques[b][0]; r <= bloques[b][1]; r++) { if (!sh.getRange(r, 1).getValue()) continue; // sin nombre, salta var ciclo = parseInt(sh.getRange(r, 37).getValue(), 10); // AK var ref = sh.getRange(r, 38).getValue(); // AL if (!ciclo || isNaN(ciclo)) continue; if (Object.prototype.toString.call(ref) !== '[object Date]' || isNaN(ref)) continue; var refBase = new Date(ref.getFullYear(), ref.getMonth(), ref.getDate()); for (var d = 1; d <= dias; d++) { var celda = sh.getRange(r, d + 2); // día 1 = columna C (3) var v = String(celda.getValue()).toUpperCase().trim(); if (v !== '' && v !== 'F') continue; // respeta marcas manuales var fecha = new Date(anio, mes - 1, d); var diff = Math.round((fecha - refBase) / DIA_MS); var resto = ((diff % ciclo) + ciclo) % ciclo; if (resto === 0) { celda.setValue('F'); puestos++; } else if (v === 'F') { celda.setValue(''); } } } } avisar('Francos generados: ' + puestos + ' marcas F colocadas según ciclo y fecha de referencia.'); } // ---------- DUPLICADOR DE MES (archivo histórico) ---------- // Copia Hoja1 completa como hoja nueva "Septiembre 2026" (queda de historial), // avanza la configuración al mes siguiente y limpia las marcas de la grilla. function archivarYAvanzarMes() { var ss = SpreadsheetApp.getActive(); var sh = ss.getSheetByName('Hoja1'); if (!sh) return; var mesTxt = String(sh.getRange('B1').getValue()); var anio = parseInt(sh.getRange('E1').getValue(), 10); var meses = ['Enero','Febrero','Marzo','Abril','Mayo','Junio','Julio', 'Agosto','Septiembre','Octubre','Noviembre','Diciembre']; var idx = meses.indexOf(mesTxt); if (idx < 0) { avisar('No reconozco el mes en B1: ' + mesTxt); return; } // 1) archivar copia con el nombre del mes var nombreHist = mesTxt + ' ' + anio; if (ss.getSheetByName(nombreHist)) nombreHist += ' (copia)'; var copia = sh.copyTo(ss).setName(nombreHist); // 2) avanzar al mes siguiente var nuevoMes, nuevoAnio = anio; if (idx === 11) { nuevoMes = meses[0]; nuevoAnio = anio + 1; } else { nuevoMes = meses[idx + 1]; } sh.getRange('B1').setValue(nuevoMes); sh.getRange('E1').setValue(nuevoAnio); // 3) limpiar marcas de la grilla (se conservan nombres, horarios y configuración) sh.getRange('C4:AG10').clearContent(); sh.getRange('C13:AG19').clearContent(); sh.getRange('C22:AG22').clearContent(); avisar('Mes archivado como "' + nombreHist + '". La grilla quedó lista para ' + nuevoMes + ' ' + nuevoAnio + '. Ahora usá "Generar francos del mes" y cargá vacaciones.'); } // ---------- REGISTRO DE ÚLTIMA MODIFICACIÓN ---------- function onEdit(e) { try { var ss = SpreadsheetApp.getActive(); var sh = ss.getSheetByName('Hoja1'); if (!sh) return; if (e && e.range && e.range.getSheet().getName() === 'Hoja1' && e.range.getA1Notation() === 'P1') { return; // no registrar la propia marca de tiempo } sh.getRange('P1').setValue(new Date()).setNumberFormat('dd/mm/yyyy hh:mm'); } catch (err) {} } function inicializarMarcaModificacion() { var sh = SpreadsheetApp.getActive().getSheetByName('Hoja1'); sh.getRange('P1').setValue(new Date()).setNumberFormat('dd/mm/yyyy hh:mm'); } // ---------- utilidad ---------- function avisar(texto) { try { SpreadsheetApp.getUi().alert(texto); } catch (e) { // sin UI disponible } }
